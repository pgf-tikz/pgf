% BOXTEST to be standardized

\csname newbox\endcsname\PGFTESTBOX% newbox is \outer in plain

\csname protected\endcsname\long\def\SHOWBOX#1{%
  \begingroup
    \long\def\use##1{##1}%
    \showboxbreadth=\maxdimen
    \showboxdepth=\maxdimen
    \tracingonline=0 %
    \errorcontextlines=-1 %
    \showbox\use{#1}%
  \endgroup
}

\csname protected\endcsname\long\def\BEGINBOXTEST#1{%
  \BEGINTEST{#1}%
    \setbox\PGFTESTBOX=\hbox\bgroup
      \ignorespaces
}

\csname protected\endcsname\long\def\ENDBOXTEST{%
      \ifhmode\unskip\fi
    \egroup
    \SHOWBOX{\PGFTESTBOX}%
  \ENDTEST
}
